27def apply_matplotlib_theme(): 

28 """设置全局样式，最重要的是中文字体。 

29 

30 这里踩过一个坑：直接把字体文件路径写进 font.sans-serif 是没用的， 

31 matplotlib 不认路径，会默默退回默认字体，然后中文全变成方框。 

32 正确做法是先用 font_manager.addfont 把字体注册进去， 

33 再按字体族名引用。 

34 """ 

35 if CHINESE_FONT: 35 ↛ 45line 35 didn't jump to line 45 because the condition on line 35 was always true

36 try: 

37 font_manager.fontManager.addfont(CHINESE_FONT) 

38 family = font_manager.FontProperties(fname=CHINESE_FONT).get_name() 

39 matplotlib.rcParams["font.sans-serif"] = [family, "Microsoft YaHei", 

40 "SimHei", "DejaVu Sans"] 

41 except (RuntimeError, ValueError, OSError): 

42 matplotlib.rcParams["font.sans-serif"] = ["Microsoft YaHei", "SimHei", 

43 "DejaVu Sans"] 

44 

45 matplotlib.rcParams["axes.unicode_minus"] = False # 负号显示成方框的问题 

46 matplotlib.rcParams["figure.facecolor"] = BACKGROUND_LIGHT 

47 matplotlib.rcParams["axes.facecolor"] = BACKGROUND_LIGHT 

48 matplotlib.rcParams["axes.edgecolor"] = "#D8DEE9"
