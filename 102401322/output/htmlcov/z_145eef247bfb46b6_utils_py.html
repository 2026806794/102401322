96class Timer: 

97 """用来统计一段代码跑了多久。 

98 

99 用法： 

100 with Timer("清洗弹幕", logger) as t: 

101 ... 

102 print(t.elapsed) 

103 """ 

104 

105 def __init__(self, label, logger=None): 

106 self.label = label 

107 self.logger = logger 

108 self.start = 0.0 

109 self.elapsed = 0.0 

110 

111 def __enter__(self): 

112 self.start = time.perf_counter() 

113 return self 

114 

115 def __exit__(self, exc_type, exc_value, traceback): 

116 self.elapsed = time.perf_counter() - self.start 

117 if self.logger: 

118 self.logger.info("%s 耗时 %.3f 秒", self.label, self.elapsed)
