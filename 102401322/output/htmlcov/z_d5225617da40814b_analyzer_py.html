1"""统计分析。 

2 

3把清洗后的弹幕变成能写进 Excel、能画图的数字。统计口径有这几项： 

4 

5 1. 每类弹幕的总数量 —— 按应用领域给弹幕打标签（一条弹幕可以属于多个领域） 

6 2. Top-N 弹幕词频 —— 整体和分领域都要 

7 3. Top-N 具体案例 —— 哪个大模型产品被提到得最多 

8 4. 态度 / 成本 / 风险 —— 作业 2.4 要求归纳的三类主流看法 

9 

10实现上注意两点： 

11 * 领域、产品、态度这些标签全部交给 Aho-Corasick 一次扫完，不用逐个别名去 in 判断； 

12 * 一条弹幕只分一次词，分出来的词同时喂给"整体词频"和"各领域词频"两个计数器， 

13 不然同一批数据要分词两遍，白浪费一半时间。 

14"""

35def build_automaton(lexicon): 

36 """把一个词典变成 AC 自动机，标签用标准名。""" 

37 patterns = [] 

38 for canonical, aliases in lexicon.items(): 

39 for alias in aliases: 

40 patterns.append((alias, canonical)) 

41 return AhoCorasick(patterns)

44def rank_items(counter, top_n=None): 

45 """把 {名称: 次数} 排成带名次的列表。 

46 

47 排序规则：次数从多到少，次数相同的按名称排，保证每次跑出来顺序都一样。 

48 """ 

49 items = sorted(counter.items(), key=lambda kv: (-kv[1], kv[0])) 

50 if top_n is not None: 

51 items = items[:top_n] 

52 

53 total = sum(counter.values()) 

54 if total == 0: 

55 total = 1 

56 

57 result = [] 

58 for index, (name, count) in enumerate(items, start=1): 

59 result.append({ 

60 "rank": index, 

61 "name": name, 

62 "count": count, 

63 "ratio": round(count * 1.0 / total, 4), 

64 }) 

65 return result

68def progress_bucket(time_seconds, duration): 

69 """把弹幕的出现时间换算成"视频进度的第几桶"。 

70 

71 这里必须按时长归一化。不归一化的话，一个 15 小时的课程视频 

72 弹幕数会碾压一个 1 分钟的视频，图表就失真了。 

73 """ 

74 if not duration or duration <= 0: 

75 return PROGRESS_BUCKETS[0] 

76 

77 ratio = float(time_seconds) / float(duration) 

78 ratio = max(0.0, min(ratio, 0.999)) 

79 return PROGRESS_BUCKETS[int(ratio * 10)]

112class DanmakuAnalyzer: 

113 """弹幕统计器。 

114 

115 records 是清洗后的弹幕列表，videos 是视频清单（可以不给， 

116 不给的话视频相关的统计会缺标题、播放量这些信息）。 

117 segmenter 可以传一个自己的分词器进去，测试的时候用。 

118 """ 

119 

120 def __init__(self, records, videos=None, config=None, segmenter=None): 

121 self.records = list(records) 

122 self.videos = list(videos or []) 

123 self.config = config or DEFAULT_ANALYSIS_CONFIG 

124 if segmenter is not None: 

125 self.segmenter = segmenter 

126 else: 

127 self.segmenter = Segmenter(min_length=self.config.min_word_length) 

128 

129 # 五类词典各建一个自动机 

130 self.product_ac = build_automaton(PRODUCT_LEXICON) 

131 self.domain_ac = build_automaton(DOMAIN_LEXICON) 

132 self.sentiment_ac = build_automaton(SENTIMENT_LEXICON) 

133 self.cost_ac = build_automaton(COST_LEXICON) 

134 self.risk_ac = build_automaton(RISK_LEXICON) 

135 

136 def analyze(self, clean_stats=None): 

137 """跑完整套统计，返回 AnalysisResult。""" 

138 result = AnalysisResult() 

139 result.meta = { 

140 "student_id": STUDENT_ID, 

141 "total_records": len(self.records), 

142 "top_n": self.config.top_n, 

143 } 

144 result.clean_stats = clean_stats or {} 

145 result.total_danmaku = len(self.records) 

146 

147 video_set = set() 

148 for record in self.records: 

149 if record.get("bvid"): 149 ↛ 148line 149 didn't jump to line 148 because the condition on line 149 was always true

150 video_set.add(record["bvid"]) 

151 result.total_videos = len(video_set) 

152 

153 log.info("开始统计：%d 条弹幕 / %d 个视频", result.total_danmaku, result.total_videos) 

154 

155 # 每个视频的时长，算进度分布要用 

156 durations = {} 

157 for video in self.videos: 

158 durations[video.get("bvid")] = video.get("duration", 0) 

159 

160 # 只有弹幕池抓全了的视频才算进度分布，原因见下面那个函数 

161 covered_videos = self._fully_covered_videos() 

162 

163 # 下面这一大段就是一次遍历，所有统计都在里面累加 

164 domain_counter = {} 

165 product_counter = {} 

166 sentiment_counter = {} 

167 cost_counter = {} 

168 risk_counter = {} 

169 keyword_counter = {} 

170 video_counter = {} 

171 progress_counter = {} 

172 word_counter = {} 

173 words_by_domain = {} 

174 words_by_aspect = {} 

175 

176 for record in self.records: 

177 text = record.get("text", "") 

178 if not text: 

179 continue 

180 

181 # 1) 领域标签（一条弹幕可能有多个） 

182 domains = self.domain_ac.find_labels(text) 

183 for domain in domains: 

184 domain_counter[domain] = domain_counter.get(domain, 0) + 1 

185 

186 # 2) 其他四类标签 

187 self._add_counts(product_counter, self.product_ac.find_labels(text)) 

188 sentiments = self.sentiment_ac.find_labels(text) 

189 costs = self.cost_ac.find_labels(text) 

190 risks = self.risk_ac.find_labels(text) 

191 self._add_counts(sentiment_counter, sentiments) 

192 self._add_counts(cost_counter, costs) 

193 self._add_counts(risk_counter, risks) 

194 

195 # 3) 关键词、视频、进度 

196 keyword = record.get("keyword") or "未标注" 

197 keyword_counter[keyword] = keyword_counter.get(keyword, 0) + 1 

198 

199 bvid = record.get("bvid") or "未知" 

200 video_counter[bvid] = video_counter.get(bvid, 0) + 1 

201 

202 if bvid in covered_videos: 

203 bucket = progress_bucket(record.get("time", 0), durations.get(bvid, 0)) 

204 progress_counter[bucket] = progress_counter.get(bucket, 0) + 1 

205 

206 # 4) 分词。只分一次，然后同时喂给整体词频和各个领域的词频 

207 tokens = self.segmenter.segment(text) 

208 for token in tokens: 

209 word_counter[token] = word_counter.get(token, 0) + 1 

210 

211 for domain in domains: 

212 if domain not in words_by_domain: 

213 words_by_domain[domain] = {} 

214 counter = words_by_domain[domain] 

215 for token in tokens: 

216 counter[token] = counter.get(token, 0) + 1 

217 

218 # 成本/风险/态度这几个维度的词频，用来画"担忧"主题的词云 

219 for label in costs: 

220 self._add_tokens(words_by_aspect, "成本-" + label, tokens) 

221 for label in risks: 

222 self._add_tokens(words_by_aspect, "风险-" + label, tokens) 

223 for label in sentiments: 

224 self._add_tokens(words_by_aspect, "态度-" + label, tokens) 

225 

226 # ---------------- 汇总成结果 ---------------- 

227 result.total_words = sum(word_counter.values()) 

228 result.category_stats = rank_items(domain_counter) 

229 result.case_ranking = rank_items(product_counter, self.config.top_n) 

230 result.sentiment = rank_items(sentiment_counter) 

231 result.cost = rank_items(cost_counter) 

232 result.risk = rank_items(risk_counter) 

233 result.keyword_stats = rank_items(keyword_counter) 

234 

235 result.top_words = [] 

236 for index, (word, count) in enumerate( 

237 self.segmenter.top_words(word_counter, self.config.top_n), start=1): 

238 result.top_words.append({"rank": index, "word": word, "count": count}) 

239 

240 result.word_freq = [] 

241 sorted_words = sorted(word_counter.items(), key=lambda kv: (-kv[1], kv[0])) 

242 for word, count in sorted_words[:500]: 

243 result.word_freq.append({"word": word, "count": count}) 

244 

245 # 分领域词频。这里刻意多存一些词（60 个），因为分领域词云需要足够的词量； 

246 # 写进 Excel 的表格会再截取前 8 个 

247 domain_order = sorted(words_by_domain.keys(), 

248 key=lambda d: -domain_counter.get(d, 0)) 

249 for domain in domain_order: 

250 words = [] 

251 top = self.segmenter.top_words(words_by_domain[domain], 

252 self.config.category_word_pool) 

253 for index, (word, count) in enumerate(top, start=1): 

254 words.append({"rank": index, "word": word, "count": count}) 

255 result.top_words_by_category[domain] = words 

256 

257 aspect_order = sorted(words_by_aspect.keys(), 

258 key=lambda a: -sum(words_by_aspect[a].values())) 

259 for aspect in aspect_order: 

260 words = [] 

261 top = self.segmenter.top_words(words_by_aspect[aspect], self.config.top_n * 4) 

262 for index, (word, count) in enumerate(top, start=1): 

263 words.append({"rank": index, "word": word, "count": count}) 

264 result.top_words_by_aspect[aspect] = words 

265 

266 result.progress_distribution = [] 

267 for bucket in PROGRESS_BUCKETS: 

268 result.progress_distribution.append({ 

269 "bucket": bucket, 

270 "count": progress_counter.get(bucket, 0), 

271 }) 

272 

273 result.video_stats = self._video_stats(video_counter) 

274 

275 log.info("统计完成：领域 %d 类，产品案例 %d 个，有效词 %d 个", 

276 len(result.category_stats), len(product_counter), result.total_words) 

277 return result

282 for key in keys: 

283 counter[key] = counter.get(key, 0) + 1

285 @staticmethod 

286 def _add_tokens(words_by_aspect, aspect, tokens): 

287 """把一条弹幕的分词结果加到某个维度的计数器里。""" 

288 if aspect not in words_by_aspect: 

289 words_by_aspect[aspect] = {} 

290 counter = words_by_aspect[aspect] 

291 for token in tokens: 

292 counter[token] = counter.get(token, 0) + 1

294 def _fully_covered_videos(self): 

295 """挑出"弹幕池基本抓全"的视频。 

296 

297 为什么需要这个：B站对超长视频的弹幕池会截断。比如一个 15 小时、 

298 1 万条弹幕的合集课，接口只返回最早的 1200 条。这些弹幕全部落在视频开头， 

299 如果不剔除，进度分布图的第一桶会被灌成一个假尖峰， 

300 看起来就像"用户只看开头"——其实只是数据被截断了。 

301 

302 判断标准是 抓到的条数 / 视频标注的总条数 >= 0.6。 

303 """ 

304 covered = set() 

305 for video in self.videos: 

306 bvid = video.get("bvid") 

307 total = video.get("danmaku_total") or 0 

308 fetched = video.get("fetched") or 0 

309 if not bvid or total < 30: 

310 continue 

311 if fetched * 1.0 / total >= 0.6: 

312 covered.add(bvid) 

313 return covered

315 def _video_stats(self, video_counter): 

316 """按弹幕条数排序的视频清单，把视频信息也合并进来。""" 

317 info_by_bvid = {} 

318 for video in self.videos: 

319 info_by_bvid[video.get("bvid")] = video 

320 

321 rows = [] 

322 for bvid, count in sorted(video_counter.items(), key=lambda kv: (-kv[1], kv[0])): 

323 info = info_by_bvid.get(bvid, {}) 

324 rows.append({ 

325 "bvid": bvid, 

326 "title": info.get("title", ""), 

327 "author": info.get("author", ""), 

328 "keyword": info.get("keyword", ""), 

329 "view": info.get("view", 0), 

330 "duration": info.get("duration", 0), 

331 "pubdate": info.get("pubdate", 0), 

332 "danmaku_clean": count, 

333 "danmaku_raw": info.get("danmaku_total", 0), 

334 }) 

335 return rows
