92 def collect_videos(self, keywords=None, per_keyword=None, refresh=False): 

93 """按关键词搜索，汇总出一个视频清单（跨关键词按 bvid 去重）。 

94 

95 这里有两个地方是踩过坑之后改的： 

96 

97 1. 清单是"合并"而不是"覆盖"。中途被限流打断后重新运行， 

98 之前搜到的视频要保留下来，不然白搜了。 

99 2. 搜索返回空页不能直接当成"没有更多结果"。被限流时接口会返回 

100 code=0 但结果是空的，我一开始直接 break，结果有个关键词只抓到 

101 36 个视频就停了。现在的做法是等一会儿重试一次，还空就跳过这页继续翻。 

102 """ 

103 keywords = tuple(keywords or self.config.keywords) 

104 per_keyword = per_keyword or self.config.videos_per_keyword 

105 

106 # 先把已有的清单读进来 

107 seen = {} 

108 if not refresh: 

109 for item in read_json(VIDEO_INDEX_FILE, []) or []: 

110 video = _video_meta_from_dict(item) 

111 seen[video.bvid] = video 

112 

113 # 看看哪些关键词还没抓够 

114 count_by_keyword = Counter(v.keyword for v in seen.values()) 

115 todo = [] 

116 for keyword in keywords: 

117 if count_by_keyword.get(keyword, 0) < per_keyword: 

118 todo.append(keyword) 

119 

120 if not todo: 

121 log.info("视频清单已经够了（%d 个），跳过搜索", len(seen)) 

122 return list(seen.values()) 

123 

124 log.info("需要补搜的关键词：%s（现有 %d 个视频）", 

125 "、".join(f"{k}({count_by_keyword.get(k, 0)})" for k in todo), 

126 len(seen)) 

127 

128 for keyword in todo: 

129 pages = math.ceil(per_keyword / self.config.page_size) 

130 collected = count_by_keyword.get(keyword, 0) 

131 empty_pages = 0 

132 

133 for page in range(1, pages + 1): 

134 items = self._search_one_page(keyword, page) 

135 

136 if not items: 

137 empty_pages += 1 

138 if empty_pages >= 2: 

139 log.warning("关键词「%s」连续 %d 页为空，停止翻页", keyword, empty_pages) 

140 break 

141 continue 

142 empty_pages = 0 

143 

144 for item in items: 

145 bvid = item["bvid"] 

146 collected += 1 

147 if bvid in seen: 

148 continue 

149 seen[bvid] = VideoMeta( 

150 bvid=bvid, 

151 title=item["title"], 

152 author=item["author"], 

153 keyword=keyword, 

154 search_rank=item["search_rank"], 

155 duration=_parse_duration(item.get("duration", 0)), 

156 pubdate=item.get("pubdate", 0), 

157 danmaku_total=item.get("danmaku_count", 0), 

158 view=item.get("play", 0), 

159 ) 

160 

161 log.info("关键词「%s」已抓取 %d/%d 个视频（第 %d/%d 页）", 

162 keyword, collected, per_keyword, page, pages) 

163 

164 if collected >= per_keyword: 

165 break 

166 

167 # 每搜完一个关键词就存一次盘，中断了也不至于全丢 

168 write_json(VIDEO_INDEX_FILE, [v.to_dict() for v in seen.values()]) 

169 

170 videos = list(seen.values()) 

171 write_json(VIDEO_INDEX_FILE, [v.to_dict() for v in videos]) 

172 log.info("视频清单汇总完成：%d 个去重后的视频", len(videos)) 

173 return videos

176 """搜一页。返回空的话等几秒再试一次（区分"真的没有"和"被限流"）。""" 

177 try: 

178 items = self.client.search_videos(keyword, page) 

179 except BilibiliAPIError as e: 

180 log.error("搜索「%s」第 %d 页失败：%s", keyword, page, e) 

181 return [] 

182 

183 if items: 

184 return items 

185 

186 wait = random.uniform(8.0, 16.0) 

187 log.info("搜索「%s」第 %d 页返回空，%.0f 秒后重试一次", keyword, page, wait) 

188 time.sleep(wait) 

189 try: 

190 return self.client.search_videos(keyword, page) 

191 except BilibiliAPIError as e: 

192 log.error("搜索「%s」第 %d 页重试也失败：%s", keyword, page, e) 

193 return []

195 def sync_index_with_cache(self): 

196 """把"缓存里有弹幕、但清单里没有"的视频补进清单。 

197 

198 为什么会出现这种情况：搜索结果会随时间变化，早期抓到的视频 

199 在后面的搜索结果里可能已经不在了，于是它留在缓存里但掉出了清单， 

200 统计的时候就没有标题、播放量这些信息。这里做一次对齐。 

201 """ 

202 videos = {} 

203 for item in read_json(VIDEO_INDEX_FILE, []) or []: 

204 video = _video_meta_from_dict(item) 

205 videos[video.bvid] = video 

206 

207 added = 0 

208 for path in sorted(self.raw_dir.glob("*.json")): 

209 bvid = path.stem 

210 if bvid in videos: 

211 continue 

212 

213 cached = read_json(path, {}) 

214 if not cached.get("danmaku"): 

215 continue # 没有弹幕数据的不用进清单 

216 

217 video = VideoMeta( 

218 bvid=bvid, 

219 title=cached.get("title", ""), 

220 author=cached.get("author", ""), 

221 keyword=cached.get("keyword", ""), 

222 duration=cached.get("duration", 0), 

223 cid=cached.get("cid"), 

224 fetched=cached.get("fetched", 0), 

225 status=cached.get("status", "ok"), 

226 ) 

227 # 尽量从接口补全信息，补不到就用缓存里的 

228 try: 

229 detail = self.client.get_video_detail(bvid) 

230 video.title = detail.get("title") or video.title 

231 video.author = detail.get("owner") or video.author 

232 video.duration = detail.get("duration", video.duration) 

233 video.pubdate = detail.get("pubdate", 0) 

234 video.tname = detail.get("tname", "") 

235 video.view = detail.get("view", 0) 

236 video.danmaku_total = detail.get("danmaku", 0) 

237 except (BilibiliAPIError, ValueError) as e: 

238 log.info("补全 %s 的信息失败，用缓存里的：%s", bvid, e) 

239 

240 videos[bvid] = video 

241 added += 1 

242 

243 if added: 

244 write_json(VIDEO_INDEX_FILE, [v.to_dict() for v in videos.values()]) 

245 log.info("清单与缓存对齐：补入 %d 个视频，现在共 %d 个", added, len(videos)) 

246 else: 

247 log.info("清单与缓存一致，不需要对齐") 

248 return added

255 def fetch_video_danmaku(self, video, force=False): 

256 """抓一个视频的弹幕，带缓存。 

257 

258 如果缓存里状态是 ok，就直接用缓存，不再请求。 

259 状态是 empty/failed 的会重新抓一次（这些多半是当时被限流了）。 

260 """ 

261 cache_file = self._cache_path(video.bvid) 

262 cached = None 

263 if self.config.use_cache and not force: 

264 cached = read_json(cache_file) 

265 

266 if cached and cached.get("status") == "ok": 

267 video.cid = cached.get("cid") 

268 video.fetched = cached.get("fetched", 0) 

269 video.source = cached.get("source", "cache") 

270 video.status = "ok" 

271 return cached.get("danmaku", []) 

272 

273 records = [] 

274 source = "" 

275 try: 

276 detail = self.client.get_video_detail(video.bvid) 

277 video.cid = detail.get("cid") 

278 video.title = detail.get("title") or video.title 

279 video.author = detail.get("owner") or video.author 

280 video.duration = detail.get("duration", video.duration) 

281 video.pubdate = detail.get("pubdate", video.pubdate) 

282 video.tname = detail.get("tname", "") 

283 video.view = detail.get("view", video.view) 

284 video.danmaku_total = detail.get("danmaku", video.danmaku_total) 

285 

286 if not video.cid: 

287 raise BilibiliAPIError(-1, "视频详情里没有 cid", video.bvid) 

288 

289 records, source = self._fetch_danmaku_by_cid( 

290 video.cid, video.duration, video.danmaku_total) 

291 

292 video.fetched = len(records) 

293 video.source = source 

294 if records: 

295 video.status = "ok" 

296 else: 

297 video.status = "empty" 

298 

299 except (BilibiliAPIError, ValueError, KeyError) as e: 

300 log.warning("视频 %s 抓取失败：%s", video.bvid, e) 

301 video.status = "failed" 

302 video.error = str(e) 

303 

304 # 失败的结果也写缓存，方便事后查；下次运行会自动重试 

305 danmaku_list = [] 

306 for record in records: 

307 item = dict(record) 

308 item["bvid"] = video.bvid 

309 item["keyword"] = video.keyword 

310 item["video_title"] = video.title 

311 item["author"] = video.author 

312 danmaku_list.append(item) 

313 

314 write_json(cache_file, { 

315 "bvid": video.bvid, 

316 "cid": video.cid, 

317 "title": video.title, 

318 "author": video.author, 

319 "keyword": video.keyword, 

320 "duration": video.duration, 

321 "source": source, 

322 "status": video.status, 

323 "fetched": video.fetched, 

324 "danmaku": danmaku_list, 

325 }) 

326 return records

328 def _fetch_danmaku_by_cid(self, cid, duration=0, expected_total=-1): 

329 """抓弹幕，三个通道依次尝试，哪个先拿到数据就用哪个。 

330 

331 为什么要搞三个通道：我一开始只用 api 的 list.so，被限流之后 

332 所有视频都返回 412，一晚上几乎没抓到东西。后来一个个接口试， 

333 发现它们的限流策略不一样： 

334 

335 1. comment.bilibili.com/{cid}.xml —— 一次请求拿全部，限流最松（主力） 

336 2. api 的 list.so —— 格式一样但限流很严，经常 412 

337 3. api 的 seg.so —— protobuf 分段（6分钟一段）， 

338 属于另一个限流桶，当兜底 

339 

340 每个通道还加了"熔断"：连续失败 3 次就认为这个通道暂时不能用了， 

341 后面的视频直接跳过它。不这么做的话，每个视频都要为一次注定失败的 

342 请求等 2 分钟退避，486 个视频就是十几个小时。 

343 

344 expected_total 是视频总共多少条弹幕。如果是 0 就直接返回—— 

345 新上传的视频经常一条弹幕都没有，不加这个判断的话每个空视频 

346 都会白跑最多 12 次分段请求。 

347 """ 

348 if expected_total == 0: 

349 return [], "" 

350 

351 limit = self.config.channel_failure_limit 

352 records = [] 

353 source = "" 

354 

355 # 前两个通道都是"一次拿全部"，拿到 100 条以上就不用再试了 

356 channels = [ 

357 ("comment", self.client.fetch_danmaku_comment_xml, "comment.xml"), 

358 ("list", self.client.fetch_danmaku_xml, "list.so"), 

359 ] 

360 for name, fetch_func, label in channels: 

361 if len(records) >= 100: 

362 break 

363 if self.channel_failures[name] >= limit: 

364 continue 

365 

366 try: 

367 got = fetch_func(cid) 

368 self.channel_failures[name] = 0 

369 if got: 

370 records = got 

371 source = label 

372 except (BilibiliAPIError, ValueError) as e: 

373 self.channel_failures[name] += 1 

374 log.info("%s 抓取失败(cid=%s，累计 %d 次)：%s", 

375 label, cid, self.channel_failures[name], e) 

376 if self.channel_failures[name] == limit: 

377 log.warning("%s 连续失败 %d 次，判定为通道级限流，后面跳过它", 

378 label, limit) 

379 

380 # 前面都没拿到（或者拿到的太少），用分段接口兜底 

381 if self._need_segment_fallback(records, expected_total): 

382 if duration and duration > 0: 382 ↛ 385line 382 didn't jump to line 385 because the condition on line 382 was always true

383 need_segments = math.ceil(duration / 360) 

384 else: 

385 need_segments = 1 

386 need_segments = min(need_segments, self.config.max_segments) 

387 

388 for index in range(1, need_segments + 1): 

389 try: 

390 records.extend(self.client.fetch_danmaku_segment(cid, index)) 

391 except (BilibiliAPIError, ValueError) as e: 

392 log.debug("分段 %d 抓取失败(cid=%s)：%s", index, cid, e) 

393 break 

394 

395 # 两条通道的数据可能有重复，按 (时间, 内容) 去个重 

396 unique = {} 

397 for record in records: 

398 key = (record.get("time", 0.0), record.get("text", "")) 

399 unique[key] = record 

400 records = list(unique.values()) 

401 

402 if records: 

403 source = "seg.so" 

404 

405 records.sort(key=lambda r: r.get("time", 0.0)) 

406 return records[:self.config.max_danmaku_per_video], source

408 @staticmethod 

409 def _need_segment_fallback(records, expected_total): 

410 """判断要不要动用分段兜底通道。 

411 

412 前面两个通道返回的是**完整的弹幕池**，所以"拿到了但数量不多" 

413 通常说明这个视频本来就没多少弹幕，再去逐段抓就是白跑十几次请求。 

414 只有两种情况值得补抓： 

415 1. 前面全都失败了，一条没拿到； 

416 2. 拿到了，但相对视频标注的总数明显偏少（不到一半，而且总数上百）， 

417 可能是接口那边截断了。 

418 """ 

419 if not records: 

420 return True 

421 if expected_total >= 100 and len(records) < expected_total * 0.5: 

422 return True 

423 return False

427 def run(self, keywords=None, per_keyword=None, video_limit=None, 

428 force=False, refresh_index=False): 

429 """完整流程：收集视频 -> 同步清单 -> 逐个抓弹幕 -> 回写清单。 

430 

431 video_limit 是只抓前 N 个视频，调试的时候用。 

432 """ 

433 start_time = time.perf_counter() 

434 self.client.bootstrap() 

435 

436 self.collect_videos(keywords, per_keyword, refresh=refresh_index) 

437 self.sync_index_with_cache() 

438 

439 # 重新读一次清单：上面两步都可能往里加过东西 

440 videos = [_video_meta_from_dict(item) 

441 for item in read_json(VIDEO_INDEX_FILE, []) or []] 

442 if video_limit: 

443 videos = videos[:video_limit] 

444 

445 report = make_progress_logger(len(videos), step=10) 

446 ok = empty = failed = 0 

447 

448 for index, video in enumerate(videos, start=1): 

449 self.fetch_video_danmaku(video, force=force) 

450 if video.status == "ok": 

451 ok += 1 

452 elif video.status == "empty": 

453 empty += 1 

454 else: 

455 failed += 1 

456 report(index) 

457 

458 write_json(VIDEO_INDEX_FILE, [v.to_dict() for v in videos]) 

459 total_danmaku = sum(v.fetched for v in videos) 

460 log.info("弹幕抓取完成：视频 %d 个（成功 %d / 空 %d / 失败 %d），" 

461 "累计弹幕 %d 条，耗时 %.1f 分钟", 

462 len(videos), ok, empty, failed, total_danmaku, 

463 (time.perf_counter() - start_time) / 60) 

464 return videos

470def _parse_duration(value): 

471 """把 "12:34" 或 "1:02:03" 这样的时长转成秒数。 

472 

473 搜索接口给的时长是字符串，视频详情接口给的是秒数，统一成秒。 

474 """ 

475 if isinstance(value, (int, float)): 

476 return int(value) 

477 if not value: 

478 return 0 

479 

480 parts = str(value).split(":") 

481 try: 

482 numbers = [int(p) for p in parts] 

483 except ValueError: 

484 return 0 

485 

486 seconds = 0 

487 for number in numbers: 

488 seconds = seconds * 60 + number 

489 return seconds

492def iter_cached_danmaku(raw_dir=DANMAKU_RAW_DIR): 

493 """遍历所有缓存文件里的弹幕，用来离线重新统计。 

494 

495 缓存文件名就是 bvid，所以即使文件里的记录没写 bvid 也能从文件名补上。 

496 """ 

497 for path in sorted(Path(raw_dir).glob("*.json")): 

498 data = read_json(path) 

499 if not data: 

500 continue 

501 

502 bvid = data.get("bvid") or path.stem 

503 keyword = data.get("keyword", "") 

504 source = data.get("source", "") 

505 

506 for record in data.get("danmaku", []): 

507 item = dict(record) 

508 item["bvid"] = record.get("bvid") or bvid 

509 item["keyword"] = record.get("keyword") or keyword 

510 # 记录抓取通道，做"弹幕随进度分布"时要区分（分段通道只覆盖开头一段） 

511 item["source"] = source 

512 yield item
