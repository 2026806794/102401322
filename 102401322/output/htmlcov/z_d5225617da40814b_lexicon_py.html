149def iter_patterns(lexicons=None): 

150 """把词典展开成 (别名, 标准名) 的列表，直接喂给 Aho-Corasick 用。 

151 

152 例： 

153 patterns = iter_patterns({"product": PRODUCT_LEXICON}) 

154 ac = AhoCorasick(patterns) 

155 """ 

156 if lexicons is None: 156 ↛ 157line 156 didn't jump to line 157 because the condition on line 156 was never true

157 lexicons = ALL_LEXICONS 

158 

159 patterns = [] 

160 for lexicon in lexicons.values(): 

161 for canonical, aliases in lexicon.items(): 

162 for alias in aliases: 

163 patterns.append((alias, canonical)) 

164 return patterns
