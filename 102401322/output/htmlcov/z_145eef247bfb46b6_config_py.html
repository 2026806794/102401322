69class CrawlerConfig: 

70 """爬虫用到的参数。 

71 

72 之所以做成一个类而不是一堆全局变量，是为了方便临时改参数： 

73 比如调试时只抓 10 个视频，就 CrawlerConfig(videos_per_keyword=10)。 

74 """ 

75 

76 def __init__(self, 

77 keywords=("大语言模型", "大模型", "LLM"), 

78 videos_per_keyword=300, 

79 page_size=20, 

80 request_interval=(1.2, 2.8), 

81 max_retries=4, 

82 timeout=20, 

83 backoff_base=2.0, 

84 block_cooldown=(25.0, 45.0), 

85 max_segments=12, 

86 channel_failure_limit=3, 

87 max_danmaku_per_video=4000, 

88 use_cache=True): 

89 self.keywords = tuple(keywords) # 搜索关键词 

90 self.videos_per_keyword = videos_per_keyword # 每个关键词抓多少个视频 

91 self.page_size = page_size # B站搜索一页返回多少条 

92 self.request_interval = request_interval # 每次请求之间随机 sleep 的区间（秒） 

93 self.max_retries = max_retries # 单个请求最多重试几次 

94 self.timeout = timeout # 请求超时（秒） 

95 self.backoff_base = backoff_base # 重试等待时间按这个底数指数增长 

96 self.block_cooldown = block_cooldown # 连续被 412 限流后的冷却时间 

97 self.max_segments = max_segments # 一个视频最多抓几个弹幕分段 

98 self.channel_failure_limit = channel_failure_limit # 某个通道连续失败几次就放弃它 

99 self.max_danmaku_per_video = max_danmaku_per_video # 单视频弹幕上限 

100 self.use_cache = use_cache # 命中缓存就不再请求 

101 

102 self.user_agent = ("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 " 

103 "(KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36") 

104 self.referer = "https://www.bilibili.com/"
